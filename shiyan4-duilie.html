实验四 循环队列

实验要求

顺序存储实现循环队列入队、出队操作。

算法思想

max=5；(rear+1)%max==front 队列满；rear==front队空；front指向队头前一位置，rear指向队尾。

循环队列C调试代码
#include <stdio.h>
#include <stdlib.h>
#define max 5
struct Queue
{
    char data[20];
    int front,rear;
};
// 初始化队列
void iniQueue(struct Queue *s)
{
    s->front=s->rear=0;
}
// 判断队列是否未满
int noFull(struct Queue *s)
{
    if((s->rear+1)%max==s->front)
        return 0;
    else return 1;
}
// 出队
void delQueue(struct Queue *s)
{
	while(s->rear!=s->front)
	{
		s->front=(s->front+1)%max;
        printf("%4c",s->data[s->front]);
	}
	printf("\n");
}
// 入队
void insertQueue(struct Queue *s)
{
	int flag;
	char x;
	while(noFull(s))
	{
		scanf("%c",&x);
        getchar();
		s->rear=(s->rear+1)%max;
        s->data[s->rear]=x;	
	}
	delQueue(s);
}
void main()
{
	struct Queue *a=(struct Queue *)malloc(sizeof(struct Queue));
    iniQueue(a);
    char x;
    insertQueue(a,x);
}

实验五 链队列

实验要求

链式存储实现链队列，带头结点，front队头指针，rear队尾指针。

算法思想

入队：在rear后新增节点；出队：删除front后继，仅剩一个节点时rear置回头结点。
链队列C调试代码
#include <stdio.h>
#include <stdlib.h>
// 数据节点
struct queue
{
    char data;
    struct queue *next;
};
// 头尾指针结构体
struct queueSS
{
    struct queue *front,*rear;
};
// 初始化带头链队列
struct queueSS *iniQueue()
{
    struct queueSS *p;
    struct queue *r;
    p=(struct queueSS *)malloc(sizeof(struct queueSS));
    p->rear=p->front=r=(struct queue *)malloc(sizeof(struct queue));
    r->next=NULL;
    return p;
}
// 批量入队
void insertQueue(struct queueSS *s,int n)
{
    struct queue *r;
    char x;
    while(n>0)
    {
        r=(struct queue *)malloc(sizeof(struct queue));
        scanf("%c",&x);
        getchar();
        r->data=x;
        r->next=NULL;
        s->rear->next=r;
        s->rear=r;
        n--;
    }
};
// 批量出队
void delQueue(struct queueSS *p,int n)
{
    struct queue *r;
    while(n>0&&p->rear!=p->front)
    {
        if(p->front->next!=p->rear)
        {
            r=p->front->next;
            printf("%4c",r->data);
            p->front->next=r->next;
            free(r);
        }
        else
        {
            r=p->front->next;
            printf("%4c",r->data);
            p->front->next=NULL;
            p->rear=p->front;
            free(r);
        }
        n--;
    }
};
void main()
{
    struct queueSS *p;
    int i;
    p=iniQueue();
    printf("输入个数：");
    scanf("%d",&i);
    insertQueue(p,i);
    delQueue(p,i);
}
