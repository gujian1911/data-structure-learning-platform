实验三 栈

一、顺序存储栈

【实验要求】顺序存储实现栈入栈、出栈
【算法思想】栈空top=-1，入栈top+1，出栈top-1，数组存储元素

顺序栈C调试代码
#include <stdio.h>
#include <stdlib.h>
//入栈函数，返回最终栈顶下标
int push(char s[],int n)
{
    char c;
    int top=-1; //栈初始为空
    while(n>0)
    {
        top++;
        scanf("%c",&c);
        getchar();//吸收回车换行
        s[top]=c;
        n--;
    }
    return top;
}
//批量出栈打印
void pop(char s[],int n)
{
    char e;
    int top=n;
    while(top!=-1)
    {
        e=s[top];
        top--;
        printf("%4c", e);
    }
}
void main()
{
    char a[20];
    int k,i;
    scanf("%d",&k);
    getchar();
    i=push(a,k);
    pop(a,i);
}

二、链式存储链栈

【实验要求】链式栈入栈出栈
【算法思想】仅在链表头部插入删除（头插法模拟栈）

链栈C调试代码
#include <stdio.h>
#include <stdlib.h>
struct stack
{
    char data;
    struct stack *next;
};
//批量建栈（头插入栈）
struct stack * push(int n)
{
    struct stack *p,*top;
    int i;
    top=(struct stack *)malloc(sizeof(struct stack));
    printf("输入第一个字符\n");
    getchar();
    scanf("%c",&top->data);
    top->next=NULL;
    getchar();
    printf("输入其余字符\n");
    for(i=1;i<n;i++)
    {
        p=(struct stack *)malloc(sizeof(struct stack));
        scanf("%c",&p->data);
        getchar();
        p->next=top;
        top=p;
    }
    return top;
}
//全部出栈释放结点
void pop(struct stack *top,int n)
{
    struct stack *p;
    if(top!=NULL)
    {
        while(top!=NULL)
        {
            printf("%4c",top->data);
            p=top;
            top=top->next;
            free(p);
        }
    }
    else
        printf("the stack is empty");
    printf("\n");
}
void main()
{
    int k;
    struct stack *top;
    printf("输入结点个数：");
    scanf("%d");
    top=push(k);
    pop(top,k);
}
