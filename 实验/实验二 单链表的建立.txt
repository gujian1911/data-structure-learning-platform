【算法思想】用尾插法建立一个有头结点的单链表。先申请一个空间结点作为头结点，之后所有申请的结点都插在链表的尾结点之后。算法的执行过程如下图所示。

typedef struct node
{
 int data;
 struct node *next;
}node;
void print(node *head)
{
 node *p;
 p=head->next;
 while(p!=NULL)
 {
  printf("%d ",p->data);
  p=p->next;
 }
}
node* search(node *head,int x)
{
 node *p,*q;
 q=head;
 p=head->next;
 while(p!=NULL&&p->data!=x)
 {
  q=p;
  p=p->next;
 }
 if(p==NULL)
  return 0;
 else
 {
  return q;
 }
}
void replace(node *head,int x,int y)
{
 node *p,*q;
 q=search(head,x);
 if(q==0)
  printf("没有你要替换的值\n");
 else
 {
  p=q->next;
  p->data=y;
 }
}

{
 node *p,*q;

  printf("没有你要删除的值\n");

  p=q->next;
  q->next=p->next;
