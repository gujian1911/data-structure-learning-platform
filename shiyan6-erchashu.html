实验六 二叉树（括号表示法建树 + 遍历）

【实验要求】用括号表示法建立二叉树，并实现先序、中序、后序（递归）与中序非递归遍历
【算法思想】建树用栈记录父结点；遍历按访问根结点的时机区分先序/中序/后序

二叉树C调试代码
#include <stdio.h>
#include <malloc.h>
#define MaxSize 100
typedef struct bnode
{
    char data;
    struct bnode *lchild,*rchild;
} BTNode;

// 根据括号字符串创建二叉树
void CreateBTree(BTNode * &bt,char *str)
{
    BTNode *St[MaxSize],*p=NULL;
    int top=-1,k,j=0;
    char ch;
    bt=NULL;
    ch=str[j];
    while (ch!='\0')
    {
        switch(ch)
        {
        case '(':top++;St[top]=p;k=1; break;
        case ')':top--;break;
        case ',':k=2; break;
        default:
            p=(BTNode *)malloc(sizeof(BTNode));
            p->data=ch;
            p->lchild=p->rchild=NULL;
            if (bt==NULL)
                bt=p;
            else
            {
                switch(k)
                {
                case 1:St[top]->lchild=p;break;
                case 2:St[top]->rchild=p;break;
                }
            }
        }
        j++;
        ch=str[j];
    }
}

// 括号表示法输出二叉树
void DispBTree(BTNode *bt)
{
    if (bt!=NULL)
    {
        printf("%c",bt->data);
        if (bt->lchild!=NULL || bt->rchild!=NULL)
        {
            printf("(");
            DispBTree(bt->lchild);
            if (bt->rchild!=NULL)
                printf(",");
            DispBTree(bt->rchild);
            printf(")");
        }
    }
}

// 先序递归
void preorder(BTNode *bt)
{
    if(bt!=NULL)
    {
        printf("%c ",bt->data);
        preorder(bt->lchild);
        preorder(bt->rchild);
    }
}

// 中序递归
void inorder(BTNode *bt)
{
    if(bt!=NULL)
    {
        inorder(bt->lchild);
        printf("%c ",bt->data);
        inorder(bt->rchild);
    }
}

// 后序递归
void postorder(BTNode *bt)
{
    if(bt!=NULL)
    {
        postorder(bt->lchild);
        postorder(bt->rchild);
        printf("%c ",bt->data);
    }
}

// 中序非递归栈实现
void inorder1(BTNode *bt)
{
    BTNode *St[MaxSize],*p=bt;
    int top=-1;
    while((p!=NULL)||(top!=-1))
    {
        while(p)
        {
            top++;
            St[top]=p;
            p=p->lchild;
        }
        if(top!=-1)
        {
            p=St[top];
            top--;
            printf("%c ",p->data);
            p=p->rchild;
        }
    }
}

// 递归求树高度
int BTHeight(BTNode *bt)
{
    if(bt==NULL) return 0;
    int lh=BTHeight(bt->lchild);
    int rh=BTHeight(bt->rchild);
    return lh>rh ? lh+1 : rh+1;
}

// 递归统计总节点
int NodeCount(BTNode *bt)
{
    if(bt==NULL) return 0;
    return NodeCount(bt->lchild)+NodeCount(bt->rchild)+1;
}

// 递归统计叶子节点
int LeafCount(BTNode *bt)
{
    if(bt==NULL) return 0;
    if(bt->lchild==NULL && bt->rchild==NULL)
        return 1;
    return LeafCount(bt->lchild)+LeafCount(bt->rchild);
}

int main()
{
    BTNode *bt;
    char s[MaxSize];
    printf("以括号表示法输入二叉树：\n");
    gets(s);
    CreateBTree(bt,s);
    printf("以括号表示法输出二叉树:\n");
    DispBTree(bt);
    printf("\n先序递归遍历:\n");
    preorder(bt);
    printf("\n中序递归遍历:\n");
    inorder(bt);
    printf("\n中序非递归遍历:\n");
    inorder1(bt);
    printf("\n后序递归遍历:\n");
    postorder(bt);
    printf("\n树高度:%d\n",BTHeight(bt));
    printf("总节点数:%d\n",NodeCount(bt));
    printf("叶子节点数:%d\n",LeafCount(bt));
    return 0;
}
