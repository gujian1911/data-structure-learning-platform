实验一 顺序表的基本运算

实验说明

本实验用C语言静态数组实现顺序表，含查找、替换、按位置/按值插入删除共6种操作； 右侧动画逐步演示，左侧代码高亮跟踪执行行，运行日志同步 printf 输出。位置从 1 开始计数。

📖 展开详细说明

本实验采用C语言静态数组实现顺序表，包含6种基础操作：元素查找、元素替换、按位置插入、按值删除、按值插入、按位置删除。 右侧网页提供可视化动画模拟每一步操作流程，运行日志同步还原C语言控制台printf输出效果。

左侧代码区会动态跟踪：执行到哪一句，那一行就高亮显示（橙色），已执行过的行留浅蓝底色。 按位置插入的「位置」从 1 开始计数（第 1 个位置即下标 0），可取 1 ~ 长度+1。

动画区按顺序表的连续存储结构绘制：蓝色矩形是已存放的元素，淡灰「空」矩形是数组提前申请、暂时空着的预留空间（MAXSIZE = 100）。 插入前要先判断 length >= MAXSIZE，空间够才把元素整体后移腾位，否则提示「顺序表已满」。

按值插入的位置计算：先用查找得到参考数值的下标 loc—— 选「前」表示插到参考值的当前位置，即第 loc + 1 个位置（元素整体后移，参考值被顶到后面）； 选「后」表示插到参考值后一个位置，即第 loc + 2 个位置。 两种选择都复用 insertByIndex 完成空间判断与后移插入。

C语言完整实验代码

#include <stdio.h>
#include <string.h>
#define MAXSIZE 100  // 数组最大容量

int flag = 0;
// 使用数组存储顺序表数据
int seqArr[MAXSIZE];
int length = 4;  // 当前顺序表有效元素长度

// 1. 查找元素，返回下标，未找到返回-1
int searchX(int x) {
    flag = 0;
    for (int i = 0; i < length; i++) {
        if (x == seqArr[i]) {
            flag = 1;
            printf("查找成功：%d的位置是%d\n", x, i);
            return i;
        }
    }
    if (flag == 0) {
        printf("未找到数值：%d\n", x);
        return -1;
    }
    return -1;
}

// 2. 替换指定数值元素
void replaceX(int x, int y) {
    int loc = searchX(x);
    if (loc == -1) {
        printf("没有可替换的数值：%d\n", x);
    } else {
        seqArr[loc] = y;
    }
    printf("替换后列表：");
    for (int i = 0; i < length; i++) printf("%d ", seqArr[i]);
    printf("\n");
}

// 3. 按位置插入元素（位置从1开始，可取 1 ~ length+1）
void insertByIndex(int pos, int x) {
    if (length >= MAXSIZE) {                  // 预留空间已用完
        printf("顺序表已满（MAXSIZE=%d），无法插入\n", MAXSIZE);
        return;
    }
    if (pos < 1) pos = 1;                     // 位置不能小于1
    if (pos > length + 1) pos = length + 1;   // 位置不能超过 length+1
    int loc = pos - 1;                        // 位置 → 下标
    // 数组元素后移腾出插入位置
    for (int j = length - 1; j >= loc; j--) {
        seqArr[j + 1] = seqArr[j];
    }
    seqArr[loc] = x;
    length++;
    printf("按位置插入后列表：");
    for (int i = 0; i < length; i++) printf("%d ", seqArr[i]);
    printf("\n");
}

// 4. 按数值删除元素
void delByValue(int x) {
    int loc = searchX(x);
    if (loc == -1) {
        printf("没有要删除的数值：%d\n", x);
    } else {
        // 数组元素前移覆盖待删除值
        for (int i = loc; i < length - 1; i++) {
            seqArr[i] = seqArr[i + 1];
        }
        length--;
    }
    printf("按值删除后列表：");
    for (int i = 0; i < length; i++) printf("%d ", seqArr[i]);
    printf("\n");
}

// 5. 按参考数值插入（前/后）
void insertByValue(int target_val, int new_val, char position[]) {
    int loc = searchX(target_val);
    if (loc == -1) {
        printf("参考数值%d不存在，按值插入失败！\n", target_val);
        return;
    }
    int insert_pos;
    if (strcmp(position, "after") == 0) {
        insert_pos = loc + 2;       // 后：参考值后面一个位置（下标 + 2）
    } else {
        insert_pos = loc + 1;       // 前：参考值当前位置（下标 + 1）
    }
    insertByIndex(insert_pos, new_val);
}

// 6. 按下标位置删除元素
void delByIndex(int loc) {
    if (loc < 0 || loc >= length) {
        printf("索引%d超出列表范围（列表长度：%d），按位置删除失败！\n", loc, length);
        return;
    }
    for (int i = loc; i < length - 1; i++) {
        seqArr[i] = seqArr[i + 1];
    }
    length--;
    printf("按位置删除后列表：");
    for (int i = 0; i < length; i++) printf("%d ", seqArr[i]);
    printf("\n");
}

// 打印分隔线
void printLine() {
    printf("------------------------------\n");
}

// 主程序：数组初始化 + 菜单循环
int main() {
    // 数组初始化顺序表初始数据 [1,2,3,4]
    seqArr[0] = 1;
    seqArr[1] = 2;
    seqArr[2] = 3;
    seqArr[3] = 4;
    length = 4;

    int choice;
    int x, y, loc, target, newv;
    char pos[10];

    printf("===== 顺序表操作工具 =====\n");
    printf("初始列表：1 2 3 4\n");
    printf("提示：输入 0 可退出程序\n\n");

    while (1) {
        printf("请输入你的选项：\n");
        printf("1 按位置插入  2 按值删除  3 替换  4 查找  5 按值插入  6 按位置删除  0 退出\n");
        scanf("%d", &choice);

        if (choice == 0) {
            printf("程序已退出，感谢使用！\n");
            break;
        }
        if (choice == 1) {
            printf("请输入插入的数值：");
            scanf("%d", &x);
            printf("请输入插入的位置（第几个位置，从1开始）：");
            scanf("%d", &loc);
            insertByIndex(loc, x);
        } else if (choice == 2) {
            printf("请输入删除的数值：");
            scanf("%d", &x);
            delByValue(x);
        } else if (choice == 3) {
            printf("请输入被替换的数值：");
            scanf("%d", &x);
            printf("请输入要替换的数值：");
            scanf("%d", &y);
            replaceX(x, y);
        } else if (choice == 4) {
            printf("请输入要查找的数值：");
            scanf("%d", &x);
            searchX(x);
        } else if (choice == 5) {
            printf("请输入参考数值（要在哪个数旁边插入）：");
            scanf("%d", &target);
            printf("请输入要插入的新数值：");
            scanf("%d", &newv);
            printf("请输入插入位置（after/before）：");
            scanf("%s", pos);
            insertByValue(target, newv, pos);
        } else if (choice == 6) {
            printf("请输入要删除的元素位置：");
            scanf("%d", &loc);
            delByIndex(loc);
        } else {
            printf("输入的选项无效！请输入 0-6 之间的数字！\n");
        }
        printLine();
    }
    return 0;
}
